cmake_minimum_required(VERSION 3.22)
project(FengYinScreenCaptureProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(Threads REQUIRED)
add_executable(PcmFrameQueueTests ../../tests/test_pcm_frame_queue.cpp)
target_link_libraries(PcmFrameQueueTests PRIVATE Threads::Threads)
add_test(NAME screen_capture_pcm_queue COMMAND PcmFrameQueueTests)
add_executable(ScreenCapturePolicyTests ../../tests/test_screen_capture_policy.cpp)
add_test(NAME screen_capture_policy COMMAND ScreenCapturePolicyTests)
if(WIN32)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_executable(FengYinScreenCaptureProbe main.cpp)
    target_compile_definitions(FengYinScreenCaptureProbe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN WINVER=0x0A00 _WIN32_WINNT=0x0A00)
    target_compile_options(FengYinScreenCaptureProbe PRIVATE /utf-8 /EHsc /W4)
    target_link_libraries(FengYinScreenCaptureProbe PRIVATE windowsapp d3d11 dxgi dxguid mfplat mfreadwrite mfuuid evr ole32 uuid avrt)
    # Separate diagnostic target: neither installed nor linked to FengYin.
    add_test(NAME screen_capture_gpu_conversion COMMAND FengYinScreenCaptureProbe --gpu-selftest)
endif()
