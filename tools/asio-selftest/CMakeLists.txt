cmake_minimum_required(VERSION 3.22)
project(FengYinAsioSelfTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options(/utf-8)
endif()
add_executable(test-tone test-tone.cpp)
if(MSVC)
    target_compile_options(test-tone PRIVATE /UNDEBUG)
else()
    target_compile_options(test-tone PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME tone_encoding COMMAND test-tone)
if(WIN32)
    add_executable(FengYinAsioSelfTest WIN32 Main.cpp)
    target_include_directories(FengYinAsioSelfTest PRIVATE "${ASIO_INCLUDE_DIR}")
    target_compile_definitions(FengYinAsioSelfTest PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(FengYinAsioSelfTest PRIVATE ole32 shell32 advapi32 user32 uuid propsys)
endif()
