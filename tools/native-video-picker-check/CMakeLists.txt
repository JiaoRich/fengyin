cmake_minimum_required(VERSION 3.24)
project(FengYinNativePickerCheck VERSION 1.1.42 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32)
    message(FATAL_ERROR "Requires the real Windows Shell file dialog")
endif()
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 9.0.2 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
juce_add_console_app(FengYinVideoChooserTests)
target_sources(FengYinVideoChooserTests PRIVATE "${ROOT}/tests/test_video_file_chooser.cpp")
target_include_directories(FengYinVideoChooserTests PRIVATE "${ROOT}/native/Source")
target_compile_definitions(FengYinVideoChooserTests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1)
target_compile_options(FengYinVideoChooserTests PRIVATE /utf-8)
target_link_libraries(FengYinVideoChooserTests PRIVATE juce::juce_gui_extra juce::juce_recommended_config_flags ole32 oleaut32 shell32 uuid uiautomationcore)
enable_testing()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/evidence")
add_test(NAME native_video_picker COMMAND FengYinVideoChooserTests)
set_tests_properties(native_video_picker PROPERTIES TIMEOUT 100 WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/evidence")
