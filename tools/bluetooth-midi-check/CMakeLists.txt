cmake_minimum_required(VERSION 3.22)
project(FengYinBluetoothCheck LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 9.0.2 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
add_executable(BluetoothCheck "${ROOT}/tests/test_bluetooth_midi.cpp" "${ROOT}/native/Source/BluetoothMidiService.cpp")
target_include_directories(BluetoothCheck PRIVATE "${ROOT}/native/Source")
target_compile_definitions(BluetoothCheck PRIVATE JUCE_USE_CURL=0)
target_compile_options(BluetoothCheck PRIVATE /utf-8 /UNDEBUG)
target_link_libraries(BluetoothCheck PRIVATE juce::juce_audio_basics juce::juce_data_structures juce::juce_events windowsapp)
enable_testing()
add_test(NAME bluetooth_backend COMMAND BluetoothCheck)
