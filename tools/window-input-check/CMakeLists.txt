cmake_minimum_required(VERSION 3.22)
project(FengYinWindowInputCheck LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options(/utf-8)
endif()
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 9.0.2 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
add_executable(FengYinWindowInputCheck ../../tests/test_compact_glow.cpp)
target_include_directories(FengYinWindowInputCheck PRIVATE ../../native/Source)
target_compile_definitions(FengYinWindowInputCheck PRIVATE JUCE_MODAL_LOOPS_PERMITTED=1)
target_link_libraries(FengYinWindowInputCheck PRIVATE juce::juce_gui_basics juce::juce_recommended_config_flags)
if(WIN32)
    target_sources(FengYinWindowInputCheck PRIVATE ../../native/Source/WindowsCompatibility.manifest)
    target_link_libraries(FengYinWindowInputCheck PRIVATE dwmapi)
endif()
enable_testing()
add_test(NAME native_input_desktop COMMAND FengYinWindowInputCheck --desktop-smoke)
set_tests_properties(native_input_desktop PROPERTIES TIMEOUT 30)
