cmake_minimum_required(VERSION 3.22)
project(FengYinKongTechniqueTest VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 9.0.2 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_console_app(FengYinKongTechniqueTest PRODUCT_NAME FengYinKongTechniqueTest)
target_sources(FengYinKongTechniqueTest PRIVATE Main.cpp)
target_include_directories(FengYinKongTechniqueTest PRIVATE ../../native/Source)
target_compile_definitions(FengYinKongTechniqueTest PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1)
target_link_libraries(FengYinKongTechniqueTest PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
if(WIN32)
  target_compile_definitions(FengYinKongTechniqueTest PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(FengYinKongTechniqueTest PRIVATE shell32)
endif()
