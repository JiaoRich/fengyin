cmake_minimum_required(VERSION 3.24)
project(FengYinMediaIntegrationCheck LANGUAGES C CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This diagnostic needs the actual Windows media stack")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(FENGYIN_PROJECT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(FENGYIN_RECORDING_ROOT "${FENGYIN_PROJECT_ROOT}/third_party/recording" CACHE PATH "Verified recording dependencies")
set(FENGYIN_VLC_ROOT "${FENGYIN_PROJECT_ROOT}/third_party/vlc" CACHE PATH "Verified VLC")
add_subdirectory("${FENGYIN_PROJECT_ROOT}/native/Recorder" recorder)
add_executable(FengYinRecorderExplicitRate WIN32
    "${FENGYIN_PROJECT_ROOT}/native/Recorder/main.cpp"
    "${FENGYIN_PROJECT_ROOT}/native/Recorder/MiniAudioRecorder.cpp")
set_target_properties(FengYinRecorderExplicitRate PROPERTIES CXX_STANDARD 17)
target_include_directories(FengYinRecorderExplicitRate PRIVATE "${FENGYIN_RECORDING_ROOT}/miniaudio")
target_compile_definitions(FengYinRecorderExplicitRate PRIVATE FENGYIN_RECORDER_EXPLICIT_RATE_PROBE=1)
target_compile_options(FengYinRecorderExplicitRate PRIVATE /utf-8 /EHa)
target_link_libraries(FengYinRecorderExplicitRate PRIVATE FengYinScreenRecorderLib)
add_executable(FengYinScreenRecorder WIN32 "${FENGYIN_PROJECT_ROOT}/native/ScreenRecorder/main.cpp")
target_compile_definitions(FengYinScreenRecorder PRIVATE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_compile_options(FengYinScreenRecorder PRIVATE /utf-8)
target_link_libraries(FengYinScreenRecorder PRIVATE windowsapp d3d11 dxgi dwmapi mfplat mfreadwrite mfuuid ole32 shell32 avrt)
add_executable(FengYinVlcTests "${FENGYIN_PROJECT_ROOT}/tests/test_vlc_playback.cpp" "${FENGYIN_PROJECT_ROOT}/native/Source/VlcPlaybackEngine.cpp")
target_include_directories(FengYinVlcTests PRIVATE "${FENGYIN_PROJECT_ROOT}/native/Source" "${FENGYIN_VLC_ROOT}/sdk/include")
target_compile_definitions(FengYinVlcTests PRIVATE FENGYIN_VLC_TEST_PCM=1)
target_compile_options(FengYinVlcTests PRIVATE /utf-8)
target_link_libraries(FengYinVlcTests PRIVATE user32 gdi32)
